35impl Fault {
36    pub fn is_external_sync_abort(&self) -> bool {
37        assert!(self != &Self::None);
38        if Cpu::is_feature_implemented(Feature::RAS) {
39            assert!(!matches!(self, Fault::SyncParity | Fault::SyncParityOnWalk));
40        }
41
42        matches!(
43            self,
44            Fault::SyncExternal
45                | Fault::SyncParity
46                | Fault::SyncExternalOnWalk
47                | Fault::SyncParityOnWalk
48        )
49    }
50}

52impl Faultable for Fault {
53    fn is_fault(&self) -> bool {
54        self != &Fault::None
55    }
