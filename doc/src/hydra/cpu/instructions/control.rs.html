22    ///
23    /// This instruction branches conditionally to a label at a PC-relative offset. This instruction provides a hint that this is not a
24    /// subroutine call or return.
25    pub fn b_cond(&mut self, imm19: u32, cond: u8) {
26        let offset = shared::sint::<21, u32>(imm19 << 2) as u64;
27        if self.condition_holds(cond) {
28            self.branch_to::<64>(self.get_PC64().wrapping_add(offset), BranchType::DIR, true);
29        } else {
30            self.branch_not_taken(BranchType::DIR, true);
31        }
32    }

61    /// **CBB&lt;cc&gt;**: Compare bytes and branch
62    ///
63    /// This instruction compares the byte values in two registers, and conditionally branches to a label at a PC-relative offset if the
64    /// condition is true. This instruction provides a hint that this is not a subroutine call or return. This instruction does not affect
65    /// the condition flags.
66    pub fn cbb_cc(&mut self, cc: u8, m: Register, H: bool, imm9: u16, t: Register) {
67        if !Cpu::is_feature_implemented(Feature::CMPBR) {
68            self.end_of_decode(DecodeType::UNDEF);
69        }
70
71        let (op, unsigned) = match cc {
72            0b000 => (CmpOp::GT, false),
73            0b001 => (CmpOp::GE, false),
74            0b010 => (CmpOp::GT, true),
75            0b011 => (CmpOp::GE, true),
76            0b110 => (CmpOp::EQ, true),
77            0b111 => (CmpOp::NE, true),
78            _ => unreachable!(),
79        };
80
81        let offset = shared::sint::<11, u16>(imm9 << 2) as u64;
82
83        let datasize = 8usize << (H as u8);
84        let operand1 = n_bits(self.x[t], datasize);
85        let operand2 = n_bits(self.x[m], datasize);
86
87        let cond = if unsigned {
88            let value1 = operand1;
89            let value2 = operand2;
90
91            match op {
92                CmpOp::EQ => value1 == value2,
93                CmpOp::NE => value1 != value2,
94                CmpOp::GT => value1 > value2,
95                CmpOp::GE => value1 >= value2,
96                _ => unreachable!(),
97            }
98        } else {
99            let value1 = operand1 as i64;
100            let value2 = operand2 as i64;
101
102            match op {
103                CmpOp::EQ => value1 == value2,
104                CmpOp::NE => value1 != value2,
105                CmpOp::GT => value1 > value2,
106                CmpOp::GE => value1 >= value2,
107                _ => unreachable!(),
108            }
109        };
110
111        if cond {
112            self.branch_to::<64>(self.get_PC64().wrapping_add(offset), BranchType::DIR, true);
113        } else {
114            self.branch_not_taken(BranchType::DIR, true);
115        }
116    }

271    /// **CCMN (immediate)**: Conditional compare negative (immediate)
272    ///
273    /// This instruction sets the value of the condition flags to the result of the comparison of a register value and a negated
274    /// immediate value if the condition is TRUE, and an immediate value otherwise.
275    ///
276    /// ```
277    /// # use hydra::cpu::{Cpu, ExtendType, common::NZCV};
278    /// # let mut cpu = Cpu::default();
279    /// cpu.__dbg_set_x(0, 0xFFFF_FFFF);
280    /// cpu.__dbg_get_pstate().__dbg_set_Z(true);  // make the condition hold
281    /// cpu.ccmn_imm(false, 1, 0b0000, 0, 0b1111);
282    /// assert_eq!(cpu.__dbg_get_pstate().nzcv(), NZCV::init(false, true, true, false));
283    /// ```
284    ///
285    /// ```
286    /// # use hydra::cpu::{Cpu, ExtendType, common::NZCV};
287    /// # let mut cpu = Cpu::default();
288    /// cpu.__dbg_set_x(0, 0xFFFF_FFFF);
289    /// cpu.__dbg_get_pstate().__dbg_set_Z(false);  // make the condition not hold
290    /// cpu.ccmn_imm(false, 1, 0b0000, 0, 0b1111);
291    /// assert_eq!(cpu.__dbg_get_pstate().nzcv(), NZCV::init(true, true, true, true));
292    /// ```
293    pub fn ccmn_imm(&mut self, sf: bool, imm5: u8, cond: u8, n: Register, nzcv: u8) {
294        let mut flags = NZCV::from(nzcv);
295
296        if self.condition_holds(cond) {
297            let operand1 = self._get_op(n, sf);
298            let operand2 = imm5 as u64;
299
300            let (_, rflags) = if sf {
301                shared::add_with_carry::<64>(operand1, operand2, false)
302            } else {
303                shared::add_with_carry::<32>(operand1, operand2, false)
304            };
305
306            flags = rflags;
307        }
308
309        self.pstate.set_nzcv(&flags);
310    }

381    /// **HLT**: Halt instruction
382    ///
383    /// This instruction can generate a Halt Instruction debug event, which causes entry into Debug state.
384    /// Within a guarded memory region, while PSTATE.BTYPE != 0b00, an HLT instruction that would cause entry into Debug
385    /// state will not generate a Branch Target exception and will cause entry into Debug state as normal. For more information, see
386    /// PSTATE.BTYPE.
387    #[allow(unused_variables, unused_assignments)]
388    pub fn hlt(&mut self, imm16: u16) {
389        let mut btype_compatible = false;
390
391        if !EDSCRField::HDE.fetch_bit(self.edscr) || !self.halting_allowed() {
392            self.end_of_decode(DecodeType::UNDEF);
393        } else if Cpu::is_feature_implemented(Feature::BTI) {
394            btype_compatible = true;
395        }
396
397        let fault = FaultRecord::no_fault();
398        self.halt(DebugHalt::HaltInstruction as u8, false, fault);
399    }
