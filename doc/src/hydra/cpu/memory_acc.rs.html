283    pub fn aarch64_s1enabled(&self, regime: Regime, acctype: AccessType) -> bool {
284        // TODO: Effective TRBLIMITR yada yada
285        if acctype == AccessType::TRBE {
286            return false;
287        }
288
289        // TODO: Effective PMBLIMITR yada yada
290        if acctype == AccessType::SPE {
291            return false;
292        }
293
294        match regime {
295            Regime::EL3 => SCTLRField::M.fetch_bit(self.sctlr_el3),
296            Regime::EL2 | Regime::EL20 => SCTLRField::M.fetch_bit(self.sctlr_el2),
297            Regime::EL10 => {
298                (!self.el2_enabled()
299                    || (!HCR_EL2Field::DC.fetch_bit(self.hcr_el2)
300                        && !HCR_EL2Field::TGE.fetch_bit(self.hcr_el2)))
301                    && SCTLRField::M.fetch_bit(self.sctlr_el1)
302            }
303            _ => unreachable!(),
304        }
305    }

315    pub fn aarch64_s1decode_tg0(&self, tg0_in: u8) -> TGx {
316        if tg0_in == 0b11 {
317            panic!("Imp def bits");
318        }
319
320        match tg0_in {
321            0b00 => TGx::_4KB,
322            0b01 => TGx::_64KB,
323            0b10 => TGx::_16KB,
324            _ => unreachable!(),
325        }
326    }

328    pub fn aarch64_s1ttw_params_el3(&self) -> S1TTWParams {
329        let tgx = self.aarch64_s1decode_tg0(TCR_EL3Field::TG0.fetch_range(self.tcr_el3));
330        let txsz = TCR_EL3Field::T0SZ.fetch_range(self.tcr_el3);
331        let ps = TCR_EL3Field::PS.fetch_range(self.tcr_el3);
332        let irgn = TCR_EL3Field::IRGN0.fetch_range(self.tcr_el3);
333        let orgn = TCR_EL3Field::ORGN0.fetch_range(self.tcr_el3);
334        let sh = TCR_EL3Field::SH0.fetch_range(self.tcr_el3);
335        let tbi = TCR_EL3Field::TBI.fetch_bit(self.tcr_el3);
336        let mair = mairtype_from_mair(self.mair_el3);
337        let skl = 0;
338        let d128 = if Cpu::is_feature_implemented(Feature::D128) {
339            TCR_EL3Field::D128.fetch_bit(self.tcr_el3)
340        } else {
341            false
342        };
343        let disch = if d128 {
344            TCR_EL3Field::DisCH0.fetch_bit(self.tcr_el3)
345        } else {
346            false
347        };
348        let pie = if d128 {
349            true
350        } else if Cpu::is_feature_implemented(Feature::S1PIE) {
351            TCR_EL3Field::PIE.fetch_bit(self.tcr_el3)
352        } else {
353            false
354        };
355        let pir = [0; 16];
356        let mair2 = [0; 8];
357        let aie = if Cpu::is_feature_implemented(Feature::AIE) {
358            TCR_EL3Field::AIE.fetch_bit(self.tcr_el3)
359        } else {
360            false
361        };
362        let wxn = SCTLRField::WXN.fetch_bit(self.sctlr_el3);
363        let ee = SCTLRField::EE.fetch_bit(self.sctlr_el3);
364        let sif = if !Cpu::is_feature_implemented(Feature::RME)
365            || Cpu::is_feature_implemented(Feature::SEL2)
366        {
367            SCR_EL3Field::SIF.fetch_bit(self.scr_el3)
368        } else {
369            false
370        };
371        let tbid = if Cpu::is_feature_implemented(Feature::PAuth) {
372            TCR_EL3Field::TBID.fetch_bit(self.tcr_el3)
373        } else {
374            false
375        };
376        let mut hpd = if Cpu::is_feature_implemented(Feature::HPDS) {
377            TCR_EL3Field::HPD.fetch_bit(self.tcr_el3)
378        } else {
379            false
380        };
381        if !hpd {
382            if aie {
383                hpd = true;
384            }
385            if pie {
386                hpd = true;
387            }
388        }
389        let ha = if Cpu::is_feature_implemented(Feature::HAF) {
390            TCR_EL3Field::HA.fetch_bit(self.tcr_el3)
391        } else {
392            false
393        };
394        let hd = if Cpu::is_feature_implemented(Feature::HAFDBS) && ha {
395            TCR_EL3Field::HD.fetch_bit(self.tcr_el3)
396        } else {
397            false
398        };
399        let ds = if [TGx::_4KB, TGx::_16KB].contains(&tgx)
400            && Cpu::is_feature_implemented(Feature::LPA2)
401            && !d128
402        {
403            TCR_EL3Field::DS.fetch_bit(self.tcr_el3)
404        } else {
405            false
406        };
407        let mtx = if Cpu::is_feature_implemented(Feature::MTE_CANONICAL_TAGS)
408            || Cpu::is_feature_implemented(Feature::MTE_NO_ADDRESS_TAGS)
409        {
410            TCR_EL3Field::MTX.fetch_bit(self.tcr_el3)
411        } else {
412            false
413        };
414        let pnch = if Cpu::is_feature_implemented(Feature::THE) && !d128 {
415            TCR_EL3Field::PnCH.fetch_bit(self.tcr_el3)
416        } else {
417            false
418        };
419        let haft = if Cpu::is_feature_implemented(Feature::HAFT) && ha {
420            TCR_EL3Field::HAFT.fetch_bit(self.tcr_el3)
421        } else {
422            false
423        };
424        // let emec = if Cpu::is_feature_implemented(Feature::MEC) {
425        //     SCT
426        // }
427
428        todo!()
429    }
