55pub enum ExtendType {
56    UXTB = 0b000,
57    UXTH = 0b001,
58    UXTW = 0b010,
59    /// LSL|UXTX
60    UXTX = 0b011,
61    SXTB = 0b100,
62    SXTH = 0b101,
63    SXTW = 0b110,
64    SXTX = 0b111,
65}
66
67impl ExtendType {
68    pub fn decode(n: u8) -> Option<Self> {
69        if n > 0b111 {
70            None
71        } else {
72            Some(match n {
73                0b000 => Self::UXTB,
74                0b001 => Self::UXTH,
75                0b010 => Self::UXTW,
76                0b011 => Self::UXTX,
77                0b100 => Self::SXTB,
78                0b101 => Self::SXTH,
79                0b110 => Self::SXTW,
80                0b111 => Self::SXTX,
81                _ => unreachable!("n not in (0b000..=0b111) but also not > 0b111?"),
82            })
83        }
84    }
85}

767    /// **ADR**: Form PC-relative address
768    ///
769    /// This instruction adds an immediate value to the PC value to form a PC-relative address, and writes the result to the
770    /// destination register.
771    pub fn adr(&mut self, immlo: u8, immhi: u32, d: Register) {
772        let immlo = to_uint::<2>(immlo as u64);
773        let immhi = to_uint::<19>(immhi as u64);
774
775        let imm = (immhi << 2) | immlo;
776        self._set_op(d, self.get_PC64().wrapping_add(imm), true);
777    }
778
779    /// **ADRP**: Form PC-relative address to 4KB page
780    ///
781    /// This instruction adds an immediate value that is shifted left by 12 bits, to the PC value to form a PC-relative address, with
782    /// the bottom 12 bits masked out, and writes the result to the destination register.
783    pub fn adrp(&mut self, immlo: u8, immhi: u32, d: Register) {
784        let immlo = to_uint::<2>(immlo as u64);
785        let immhi = to_uint::<19>(immhi as u64);
786
787        let imm = ((immhi << 2) | immlo) << 12;
788        let base = self.get_PC64() & (!0u64 << 12);
789
790        self._set_op(d, base.wrapping_add(imm), true);
791    }

793    /// **AXFLAG**: Convert floating-point condition flags from Arm to external format
794    ///
795    /// This instruction converts the state of the PSTATE.{N,Z,C,V} flags from a form representing the result of an Arm
796    /// floating-point scalar compare instruction to an alternative representation required by some software.
797    pub fn axflag(&mut self) {
798        let z = self.pstate.Z() || self.pstate.V();
799        let c = self.pstate.C() && !self.pstate.V();
800
801        self.pstate.set_nzcv(&NZCV::init(false, z, c, false));
802    }

804    fn execute_instruction(&mut self, instr: Instruction) {
805        match instr {
806            Instruction::Abs { sf, rn, rd } => self.abs(sf, rn, rd),
807            Instruction::Adc { sf, rm, rn, rd } => self.adc(sf, rm, rn, rd),
808            Instruction::Adcs { sf, rm, rn, rd } => self.adcs(sf, rm, rn, rd),
809            Instruction::AddEr {
810                sf,
811                rm,
812                option,
813                imm3,
814                rn,
815                rd,
816            } => self.add_er(sf, rm, option, imm3, rn, rd),
817            Instruction::AddImm {
818                sf,
819                sh,
820                imm12,
821                rn,
822                rd,
823            } => self.add_imm(sf, sh, imm12, rn, rd),
824            Instruction::AddSr {
825                sf,
826                shift,
827                rm,
828                imm6,
829                rn,
830                rd,
831            } => self.add_sr(sf, shift, rm, imm6, rn, rd),
832            Instruction::BCond { imm19, cond } => self.b_cond(imm19, cond),
833            Instruction::Movz { sf, rd, imm16, hw } => self.movz(sf, hw, imm16, rd),
834            Instruction::SubEr {
835                sf,
836                rm,
837                option,
838                imm3,
839                rn,
840                rd,
841            } => self.sub_er(sf, rm, option, imm3, rn, rd),
842            Instruction::SubImm {
843                sf,
844                sh,
845                imm12,
846                rn,
847                rd,
848            } => self.sub_imm(sf, sh, imm12, rn, rd),
849            Instruction::SubSr {
850                sf,
851                shift,
852                rm,
853                imm6,
854                rn,
855                rd,
856            } => self.sub_sr(sf, shift, rm, imm6, rn, rd),
857            Instruction::SubsEr {
858                sf,
859                rm,
860                option,
861                imm3,
862                rn,
863                rd,
864            } => self.subs_er(sf, rm, option, imm3, rn, rd),
865            Instruction::SubsImm {
866                sf,
867                sh,
868                imm12,
869                rn,
870                rd,
871            } => self.subs_imm(sf, sh, imm12, rn, rd),
872            Instruction::SubsSr {
873                sf,
874                shift,
875                rm,
876                imm6,
877                rn,
878                rd,
879            } => self.subs_sr(sf, shift, rm, imm6, rn, rd),
880        }
881    }

883    pub fn step(&mut self, memory: &mut Memory) -> Result<(), CpuError> {
884        let instruction_word = memory
885            .read::<u32>(self.get_PC64() as usize)
886            .ok_or(CpuError::InstructionReadError)?;
887
888        let decoded = decode::decode(instruction_word);
889        if let Some(instr) = decoded {
890            println!("Executing instruction: {:?}", instr);
891            self.execute_instruction(instr);
892        }
893
894        self._pc += 4;
895
896        Ok(())
897    }
