162#[allow(non_camel_case_types)]
163pub enum OSLSR_EL1Field {
164    nTT = 2,
165    OSLK = 1,
166    OSLM = -1,
167}
168
169impl RegisterFieldRange<u64, u8> for OSLSR_EL1Field {
170    fn fetch_range(self, v: u64) -> u8 {
171        if let Self::OSLM = self {
172            return ((nth_bit(v, 3) as u8) << 1) | (nth_bit(v, 0) as u8);
173        }
174
175        unreachable!()
176    }
177}

179macro_rules! register_field {
180    ($($ty:ty $( {$value_type:ty} )?),* $(,)?) => {
181        $(
182            impl RegisterField<register_field!(@type $($value_type)?)> for $ty {
183                fn fetch_bit(self, v: register_field!(@type $($value_type)?)) -> bool {
184                    nth_bit(v, self as usize)
185                }
186            }
187        )*
188    };
189
190    (@type $ty:ty) => { $ty };
191    (@type) => { u32 };
192}
