83impl RegisterField<u64> for SCTLRField {
84    fn fetch_bit(self, sctlr_v: SCTLRType) -> bool {
85        nth_bit(sctlr_v, self as usize)
86    }
87}

89impl RegisterFieldRange<u64, u8> for SCTLRField {
90    fn fetch_range(self, sctlr_v: SCTLRType) -> u8 {
91        if let Self::TCF = self {
92            return ((sctlr_v >> 40) & 0b11) as u8;
93        }
94
95        unreachable!("??")
96    }
97}
