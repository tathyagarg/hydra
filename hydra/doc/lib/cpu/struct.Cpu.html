pub fn abs(&mut self, sf: bool, d: usize, n: usize)

ABS: Absolute value

This instruction calculates the absolute value of the signed integer value in the source register, and writes the result to the destination register. If the value in the source register is the most negative value, the result is the same as the value in the source register.

cpu.__dbg_set_x(0, -5i64 as u64);
cpu.abs(true, 1, 0);  // `ABS X1, X0`
assert_eq!(cpu.__dbg_get_x(1), 5);
cpu.__dbg_set_x(0, 9223372032559808513u64);
cpu.abs(false, 1, 0);  // `ABS W1, W0`
assert_eq!(cpu.__dbg_get_x(1), 1);

pub fn adc(&mut self, sf: bool, d: usize, n: usize, m: usize)

ADC: Add with carry

This instruction adds two register values and the Carry flag value, and writes the result to the destination register.

cpu.__dbg_set_x(0, 1);
cpu.__dbg_set_x(1, 2);
cpu.adc(true, 2, 0, 1);  // `ADC X2, X0, X1`
assert_eq!(cpu.__dbg_get_x(2), 3);
cpu.__dbg_set_x(0, 1);
cpu.__dbg_set_x(1, 2);
cpu.__dbg_get_pstate().__dbg_set_C(true);  // Set PSTATE.C = 1
cpu.adc(true, 2, 0, 1);  // `ADC X2, X0, X1`
assert_eq!(cpu.__dbg_get_x(2), 4);

pub fn adcs(&mut self, sf: bool, d: usize, n: usize, m: usize)

ADCS: Add with carry, setting flags

This instruction adds two register values and the Carry flag value, and writes the result to the destination register. It updates the condition flags based on the result.

cpu.__dbg_set_x(0, 1);
cpu.__dbg_set_x(1, 2);
cpu.adcs(true, 2, 0, 1);  // `ADCS X2, X0, X1`
assert_eq!(cpu.__dbg_get_x(2), 3);
assert!(!cpu.__dbg_get_pstate().__dbg_get_C());
cpu.__dbg_set_x(0, u64::MAX);
cpu.__dbg_set_x(1, 1);
cpu.adcs(true, 2, 0, 1);  // `ADCS X2, X0, X1`
assert_eq!(cpu.__dbg_get_x(2), 0);
assert!(cpu.__dbg_get_pstate().__dbg_get_C());

pub fn _add_er( &mut self, sf: bool, m: usize, option: u8, imm3: u8, n: RegisterSP, d: RegisterSP, )

ADD (extended register): Add extended and scaled register

This instruction adds a register value and a sign-extended or zero-extended register value, followed by an optional left shift amount, and writes the result to the destination register. The argument that is extended from the <Rm> register can be a byte, halfword, word, or doubleword.

cpu.__dbg_set_x(0, 1);
cpu.__dbg_set_x(1, 2);
// ADD X2, X0, X1, UXTB #3
cpu._add_er(true, 1, 0b000, 3, RegisterSP::Register(0), RegisterSP::Register(2));
assert_eq!(cpu.__dbg_get_x(2), 17);

We get 17 because 17 = 1 + (2 << 3) The ExtendType is not very consequential because we are dealing with such small values.

cpu.__dbg_set_x(0, 1);
cpu.__dbg_set_x(1, 500);
cpu._add_er(true, 1, 0b000, 3, RegisterSP::Register(0), RegisterSP::Register(2));
assert_eq!(cpu.__dbg_get_x(2), 1953);

We get 1953 because 1953 = 1 + (244 << 3), where 244 = 500 & 0xFF (due to UXTB extending)

cpu.__dbg_set_x(0, 1);
cpu.__dbg_set_x(1, 500);
cpu._add_er(true, 1, 0b011, 3, RegisterSP::Register(0), RegisterSP::Register(2));
assert_eq!(cpu.__dbg_get_x(2), 4001);

Without the masking to byte-width, we get 4001 instead.
