46impl ExtendType {
47    pub fn decode(n: u8) -> Option<Self> {
48        if n > 0b111 {
49            None
50        } else {
51            Some(match n {
52                0b000 => Self::UXTB,
53                0b001 => Self::UXTH,
54                0b010 => Self::UXTW,
55                0b011 => Self::UXTX,
56                0b100 => Self::SXTB,
57                0b101 => Self::SXTH,
58                0b110 => Self::SXTW,
59                0b111 => Self::SXTX,
60                _ => unreachable!("n not in (0b000..=0b111) but also not > 0b111?"),
61            })
62        }
63    }
64}

571    /// **ADR**: Form PC-relative address
572    ///
573    /// This instruction adds an immediate value to the PC value to form a PC-relative address, and writes the result to the
574    /// destination register.
575    pub fn adr(&mut self, immlo: u8, immhi: u32, d: usize) {
576        let immlo = to_uint::<2>(immlo as u64);
577        let immhi = to_uint::<19>(immhi as u64);
578
579        let imm = (immhi << 2) | immlo;
580        self.x[d] = self.get_PC64() + imm;
581    }
582
583    /// **ADRP**: Form PC-relative address to 4KB page
584    ///
585    /// This instruction adds an immediate value that is shifted left by 12 bits, to the PC value to form a PC-relative address, with
586    /// the bottom 12 bits masked out, and writes the result to the destination register.
587    pub fn adrp(&mut self, immlo: u8, immhi: u32, d: usize) {
588        let immlo = to_uint::<2>(immlo as u64);
589        let immhi = to_uint::<19>(immhi as u64);
590
591        let imm = ((immhi << 2) | immlo) << 12;
592        let base = self.get_PC64() & (!0u64 << 12);
593
594        self.x[d] = base + imm;
595    }

906    /// **CAS, CASA, CASAL, CASL**: Compare and swap word or doubleword in memory
907    ///
908    /// This instruction reads a 32-bit word or 64-bit doubleword from memory, and compares it against the value held in a first
909    /// register. If the comparison is equal, the value in a second register is written to memory. If the comparison is not equal, the
910    /// architecture permits writing the value read from the location to memory. If the write is performed, the read and write occur
911    /// atomically such that no other modification of the memory location can take place between the read and write.
912    /// - If the destination register is not one of WZR or XZR, CASA and CASAL load from memory with acquire semantics.
913    /// - CASL and CASAL store to memory with release semantics.
914    /// - CAS has neither acquire nor release semantics.
915    ///
916    /// The architecture permits that the data read clears any exclusive monitors associated with that location, even if the compare
917    /// subsequently fails.
918    ///
919    /// If the instruction generates a synchronous Data Abort, the register which is compared and loaded, that is &lt;Ws&gt;, or &lt;Xs&gt;, is
920    /// restored to the value held in the register before the instruction was executed.
921    ///
922    /// For a CAS or CASA instruction, when &lt;Ws&gt; or &lt;Xs&gt; specifies the same register as &lt;Wt&gt; or &lt;Xt&gt;, this signals to the memory
923    /// system that an additional subsequent CAS, CASA, CASAL, or CASL access to the specified location is expected to occur in the
924    /// near future in the absence of taking an exception or exceptional software conditions. The memory system can respond by
925    /// taking actions that are expected to enable the subsequent CAS, CASA, CASAL, or CASL access to succeed when it does occur.
926    ///
927    /// A code sequence starting with a CAS or CASA instruction for which &lt;Ws&gt; or &lt;Xs&gt; specifies the same register as &lt;Wt&gt; or
928    /// &lt;Xt&gt;, and ending with a subsequent CAS, CASA, CASAL, or CASL to the same location, exhibits the following properties for
929    /// best performance when the location may be accessed concurrently, on one or more other PEs:
930    /// - The sequence does not contain any direct System register writes, address translation instructions, cache or TLB
931    /// maintenance operations, exception producing instructions, exception returns, or ISB barriers.
932    /// - The execution of the sequence includes 32 or fewer instructions.
933    /// - The value provided in &lt;Ws&gt; or &lt;Xs&gt; of the first CAS or CASA is a value likely to result in the comparison failing. A
934    /// failing comparison result may lead to better performance due to the hardware not performing a write to memory.
