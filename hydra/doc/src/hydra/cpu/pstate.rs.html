21    pub(super) PACM: bool,   // PAC instruction modifier
22    pub(super) ZA: bool,     // Accumulation array enabled [SME]
23    pub(super) SM: bool,     // Streaming SVE mode enabled [SME]
24    pub(super) ALLINT: bool, // Interrupt mask bit
25    pub(super) UINJ: bool,   // Undefined Exception Injection
26    pub(super) SS: bool,     // Software step bit
27    pub(super) IL: bool,     // Illegal Execution state bit
28    pub(super) EL: u8,       // Exception level
29    pub(super) nRW: bool,    // Execution state: 0=AArch64, 1=AArch32
30    pub(super) SP: bool,     // Stack pointer select: 0=SP0, 1=SPx [AArch64 only]
31    pub(super) SSBS: bool,   // Speculative Store Bypass Safe
