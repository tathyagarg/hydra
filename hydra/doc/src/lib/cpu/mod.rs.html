273impl Cpu {
274    fn _to_op(&self, n: usize, sf: bool) -> u64 {
275        if sf {
276            self.x[n]
277        } else {
278            self.x[n] as u32 as u64
279        }
280    }
281
282    pub fn __dbg_set_x(&mut self, n: usize, v: u64) {
283        self.x[n] = v;
284    }
285
286    pub fn __dbg_get_x(&self, n: usize) -> u64 {
287        self.x[n]
288    }
289
290    /// **ABS**: Absolute value
291    ///
292    /// This instruction calculates the absolute value of the signed integer value in the source register, and writes the result to the
293    /// destination register. If the value in the source register is the most negative value, the result is the same as the value in the
294    /// source register.
295    ///
296    /// ```
297    /// # use hydra::cpu::Cpu;
298    /// # let mut cpu = Cpu::default();
299    /// cpu.__dbg_set_x(0, -5i64 as u64);
300    /// // Emulating `ABS X1, X0`
301    /// cpu.abs(true, 1, 0);
302    /// assert_eq!(cpu.__dbg_get_x(1), 5);
303    /// ```
304    pub fn abs(&mut self, sf: bool, d: usize, n: usize) {
305        let operand1 = self._to_op(n, sf) as i64;
306
307        let result = operand1.abs() as u64;
308        self.x[d] = result;
309    }
310
311    fn adc(&mut self, sf: bool, d: usize, n: usize, m: usize) {
312        let operand1 = self._to_op(n, sf);
313        let operand2 = self._to_op(m, sf);
314
315        let (result, _) = if sf {
316            shared::add_with_carry::<64>(operand1, operand2, self.pstate.C)
317        } else {
318            shared::add_with_carry::<32>(operand1, operand2, self.pstate.C)
319        };
320
321        self.x[d] = result;
322    }
323
324    fn adcs(&mut self, sf: bool, d: usize, n: usize, m: usize) {
325        let operand1 = self._to_op(n, sf);
326        let operand2 = self._to_op(m, sf);
327
328        let (result, nzcv) = if sf {
329            shared::add_with_carry::<64>(operand1, operand2, self.pstate.C)
330        } else {
331            shared::add_with_carry::<32>(operand1, operand2, self.pstate.C)
332        };
333
334        self.x[d] = result;
335        nzcv.set(&mut self.pstate);
336    }
337
338    fn add(
339        &mut self,
340        sf: bool,
341        d: RegisterSP,
342        n: RegisterSP,
343        m: usize,
344        extend_type: Option<ExtendType>,
345        amount: Option<u8>,
346    ) {
347        let shift = amount.unwrap_or(0);
348
349        let operand1 = if let RegisterSP::Register(rn) = n {
350            self.x[rn]
351        } else {
352            self.sp
353        };
354
355        let (result, _) = if sf {
356            let operand2 = shared::extend_reg::<64>(
357                m,
358                extend_type.unwrap_or(ExtendType::UXTW),
359                shift,
360                &self.x,
361            );
362            shared::add_with_carry::<64>(operand1, operand2, false)
363        } else {
364            let operand2 = shared::extend_reg::<32>(
365                m,
366                extend_type.unwrap_or(ExtendType::UXTW),
367                shift,
368                &self.x,
369            );
370            shared::add_with_carry::<32>(operand1, operand2, false)
371        };
372
373        if let RegisterSP::Register(rd) = d {
374            self.x[rd] = result;
375        } else {
376            self.sp = result;
377        }
378    }
