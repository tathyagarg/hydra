13pub enum ExtendType {
14    UXTB = 0b000,
15    UXTH = 0b001,
16    UXTW = 0b010,
17    /// LSL|UXTX
18    UXTX = 0b011,
19    SXTB = 0b100,
20    SXTH = 0b101,
21    SXTW = 0b110,
22    SXTX = 0b111,
23}
24
25impl ExtendType {
26    pub fn decode(n: u8) -> Option<Self> {
27        if n > 0b111 {
28            None
29        } else {
30            Some(match n {
31                0b000 => Self::UXTB,
32                0b001 => Self::UXTH,
33                0b010 => Self::UXTW,
34                0b011 => Self::UXTX,
35                0b100 => Self::SXTB,
36                0b101 => Self::SXTH,
37                0b110 => Self::SXTW,
38                0b111 => Self::SXTX,
39                _ => unreachable!("n not in (0b000..=0b111) but also not > 0b111?"),
40            })
41        }
42    }
43}

293impl Cpu {
294    fn _to_op(&self, n: usize, cond: bool) -> u64 {
295        if cond {
296            self.x[n]
297        } else {
298            self.x[n] as u32 as u64
299        }
300    }

302    /// A function for purely debug/testing purposes to arbitrarily set the value
303    /// of a X register to a provided v value
304    pub fn __dbg_set_x(&mut self, n: usize, v: u64) {

320    /// **ABS**: Absolute value
321    ///
322    /// This instruction calculates the absolute value of the signed integer value in the source register, and writes the result to the
323    /// destination register. If the value in the source register is the most negative value, the result is the same as the value in the
324    /// source register.
325    ///
326    /// ```
327    /// # use hydra::cpu::Cpu;
328    /// # let mut cpu = Cpu::default();
329    /// cpu.__dbg_set_x(0, -5i64 as u64);
330    /// cpu.abs(true, 0, 1);  // `ABS X1, X0`
331    /// assert_eq!(cpu.__dbg_get_x(1), 5);
332    /// ```
333    ///
334    /// ```
335    /// # use hydra::cpu::Cpu;
336    /// # let mut cpu = Cpu::default();
337    /// cpu.__dbg_set_x(0, 9223372032559808513u64);
338    /// cpu.abs(false, 0, 1);  // `ABS W1, W0`
339    /// assert_eq!(cpu.__dbg_get_x(1), 1);
340    /// ```
341    pub fn abs(&mut self, sf: bool, n: usize, d: usize) {
342        let operand1 = self._to_op(n, sf) as i64;
343
344        let result = operand1.abs() as u64;
345        self.x[d] = result;
346    }
347
348    /// **ADC**: Add with carry
349    ///
350    /// This instruction adds two register values and the Carry flag value, and writes the result to the destination register.
351    ///
352    /// ```
353    /// # use hydra::cpu::Cpu;
354    /// # let mut cpu = Cpu::default();
355    /// cpu.__dbg_set_x(0, 1);
356    /// cpu.__dbg_set_x(1, 2);
357    /// cpu.adc(true, 1, 0, 2);  // `ADC X2, X0, X1`
358    /// assert_eq!(cpu.__dbg_get_x(2), 3);
359    /// ```
360    ///
361    /// ```
362    /// # use hydra::cpu::Cpu;
363    /// # let mut cpu = Cpu::default();
364    /// cpu.__dbg_set_x(0, 1);
365    /// cpu.__dbg_set_x(1, 2);
366    /// cpu.__dbg_get_pstate().__dbg_set_C(true);  // Set PSTATE.C = 1
367    /// cpu.adc(true, 1, 0, 2);  // `ADC X2, X0, X1`
368    /// assert_eq!(cpu.__dbg_get_x(2), 4);
369    /// ```
370    pub fn adc(&mut self, sf: bool, m: usize, n: usize, d: usize) {
371        let operand1 = self._to_op(n, sf);
372        let operand2 = self._to_op(m, sf);
373
374        let (result, _) = if sf {
375            shared::add_with_carry::<64>(operand1, operand2, self.pstate.C)
376        } else {
377            shared::add_with_carry::<32>(operand1, operand2, self.pstate.C)
378        };
379
380        self.x[d] = result;
381    }
382
383    /// **ADCS**: Add with carry, setting flags
384    ///
385    /// This instruction adds two register values and the Carry flag value, and writes the result to the destination register. It updates
386    /// the condition flags based on the result.
387    ///
388    /// ```
389    /// # use hydra::cpu::Cpu;
390    /// # let mut cpu = Cpu::default();
391    /// cpu.__dbg_set_x(0, 1);
392    /// cpu.__dbg_set_x(1, 2);
393    /// cpu.adcs(true, 1, 0, 2);  // `ADCS X2, X0, X1`
394    /// assert_eq!(cpu.__dbg_get_x(2), 3);
395    /// assert!(!cpu.__dbg_get_pstate().__dbg_get_C());
396    /// ```
397    ///
398    /// ```
399    /// # use hydra::cpu::Cpu;
400    /// # let mut cpu = Cpu::default();
401    /// cpu.__dbg_set_x(0, u64::MAX);
402    /// cpu.__dbg_set_x(1, 1);
403    /// cpu.adcs(true, 1, 0, 2);  // `ADCS X2, X0, X1`
404    /// assert_eq!(cpu.__dbg_get_x(2), 0);
405    /// assert!(cpu.__dbg_get_pstate().__dbg_get_C());
406    /// ```
407    pub fn adcs(&mut self, sf: bool, m: usize, n: usize, d: usize) {
408        let operand1 = self._to_op(n, sf);
409        let operand2 = self._to_op(m, sf);
410
411        let (result, nzcv) = if sf {
412            shared::add_with_carry::<64>(operand1, operand2, self.pstate.C)
413        } else {
414            shared::add_with_carry::<32>(operand1, operand2, self.pstate.C)
415        };
416
417        self.x[d] = result;
418        nzcv.set(&mut self.pstate);
419    }

421    /// **ADD (extended register)**: Add extended and scaled register
422    ///
423    /// This instruction adds a register value and a sign-extended or zero-extended register value, followed by an optional left shift
424    /// amount, and writes the result to the destination register. The argument that is extended from the &lt;Rm&gt; register can be a byte,
425    /// halfword, word, or doubleword.
426    ///
427    /// ```
428    /// # use hydra::cpu::{Cpu, RegisterSP, ExtendType};
429    /// # let mut cpu = Cpu::default();
430    /// cpu.__dbg_set_x(0, 1);
431    /// cpu.__dbg_set_x(1, 2);
432    /// // `ADD X2, X0, X1, UXTB #3`
433    /// cpu._add_er(true, 1, 0b000, 3, RegisterSP::Register(0), RegisterSP::Register(2));
434    /// assert_eq!(cpu.__dbg_get_x(2), 17);
435    /// ```
436    ///
437    /// We get 17 because 17 = 1 + (2 << 3)
438    /// The ExtendType is not very consequential because we are dealing with such small values.
439    ///
440    /// ```
441    /// # use hydra::cpu::{Cpu, RegisterSP, ExtendType};
442    /// # let mut cpu = Cpu::default();
443    /// cpu.__dbg_set_x(0, 1);
444    /// cpu.__dbg_set_x(1, 500);
445    /// // `ADD X2, X0, X1, UXTB #3`
446    /// cpu._add_er(true, 1, 0b000, 3, RegisterSP::Register(0), RegisterSP::Register(2));
447    /// assert_eq!(cpu.__dbg_get_x(2), 1953);
448    /// ```
449    ///
450    /// We get 1953 because 1953 = 1 + (244 << 3), where 244 = 500 & 0xFF (due to UXTB extending)
451    ///
452    /// ```
453    /// # use hydra::cpu::{Cpu, RegisterSP, ExtendType};
454    /// # let mut cpu = Cpu::default();
455    /// cpu.__dbg_set_x(0, 1);
456    /// cpu.__dbg_set_x(1, 500);
457    /// // `ADD X2, X0, X1, UXTD #3`
458    /// cpu._add_er(true, 1, 0b011, 3, RegisterSP::Register(0), RegisterSP::Register(2));
459    /// assert_eq!(cpu.__dbg_get_x(2), 4001);
460    /// ```
461    ///
462    /// Without the masking to byte-width, we get 4001 instead.
463    pub fn _add_er(
464        &mut self,
465        sf: bool,
466        m: usize,
467        option: u8,
468        imm3: u8,
469        n: RegisterSP,
470        d: RegisterSP,
471    ) {
472        if imm3 > 4 {
473            panic!("imm3 too large");
474        }
475
476        let operand1 = if let RegisterSP::Register(rn) = n {
477            self.x[rn]
478        } else {
479            self.sp
480        };
481
482        let mval = self._to_op(m, option & 0b11 == 0b11);
483        let extend_type = ExtendType::decode(option).unwrap();
484
485        let (result, _) = if sf {
486            let operand2 = shared::extend_reg::<64>(mval, extend_type, imm3);
487            shared::add_with_carry::<64>(operand1, operand2, false)
488        } else {
489            let operand2 = shared::extend_reg::<32>(mval, extend_type, imm3);
490            shared::add_with_carry::<32>(operand1, operand2, false)
491        };
492
493        if let RegisterSP::Register(rd) = d {
494            self.x[rd] = result;
495        } else {
496            self.sp = result;
497        }
498    }
