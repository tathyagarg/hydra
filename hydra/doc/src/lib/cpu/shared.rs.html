14/// Signed Integer function: Converts unsigned int to signed int
15fn sint<const N: usize>(n: u64) -> i64 {
16    let mask = if N == 64 { u64::MAX } else { (1u64 << N) - 1 };
17    let sign_bit = 1u64 << (N - 1);
18    (if n & sign_bit != 0 {
19        n | !mask
20    } else {
21        n & mask
22    }) as i64
23}

25pub(super) fn add_with_carry<const N: usize>(x: u64, y: u64, carry_in: bool) -> (u64, NZCV) {
26    let mask = if N == 64 { u64::MAX } else { (1u64 << N) - 1 };
27
28    let (unsigned_sum, carry) = {
29        let (temp, c1) = x.overflowing_add(y);
30        println!("temp: {}, c1: {}", temp, c1);
31        let (temp2, c2) = temp.overflowing_add(carry_in as u64);
32        (temp2, c1 || c2)
33    };
34    let signed_sum = sint::<N>(x) + sint::<N>(y) + carry_in as i64;
35
36    let result = unsigned_sum & mask;
37
38    let n = (result >> (N - 1)) & 1 == 1;
39    let z = result == 0;
40    let c = carry;
41    let v = result as i64 != signed_sum;

49fn extend(val: u64, nbits: usize, unsigned: bool) -> u64 {
50    let mask = if nbits == 64 {
51        u64::MAX
52    } else {
53        (1u64 << nbits) - 1
54    };
55
56    let value = val & mask;
57
58    if unsigned || nbits == 64 {
59        value
60    } else {
61        let sign_bit = 1u64 << (nbits - 1);
62
63        if value & sign_bit != 0 {
64            value | !mask
65        } else {
66            value
67        }
68    }
69}

71pub(super) fn extend_reg<const N: usize>(val: u64, exttype: ExtendType, shift: u8) -> u64 {
72    let (unsigned, len): (bool, usize) = match exttype {
73        ExtendType::SXTB => (false, 8),
74        ExtendType::SXTH => (false, 16),
75        ExtendType::SXTW => (false, 32),
76        ExtendType::SXTX => (false, 64),
77        ExtendType::UXTB => (true, 8),
78        ExtendType::UXTH => (true, 16),
79        ExtendType::UXTW => (true, 32),
80        ExtendType::UXTX => (true, 64),
81    };
82
83    let nbits = len.min(N);
84    let extval = extend(val, nbits, unsigned);
85
86    return extval << shift;
87}

89pub(super) fn shift_reg(val: u64, shift_type: ShiftType, amount: usize) -> u64 {
90    match shift_type {
91        ShiftType::LSL => val << amount,
92        ShiftType::LSR => val >> amount,
93        ShiftType::ASR => (val as i64 >> amount) as u64,
94        ShiftType::ROR => val,
95    }

98fn replicate(val: u64, elem_bits: u32, total_bits: u32) -> u64 {

122fn ror(val: u64, amount: u32, width: u32) -> u64 {
123    let mask = ones(width);
124    let val = val & mask;
125    let amount = (amount % width) as u64;
126
127    if amount == 0 {
128        val
129    } else {
130        ((val >> amount) | (val << (width as u64 - amount))) & mask
131    }
132}

136/// Decode AArch64 bitfield and logical immediate masks which use a similar encoding structure
137pub(super) fn decode_bit_masks<const M: usize>(imm_n: bool, imms: u8, immr: u8) -> (u64, u64) {
138    let combined = ((imm_n as u8) << 6) | ((!imms) & 0x3f);
139    let len = 63 - (combined as u64).leading_zeros() as usize;
140    let levels = (1u8 << len) - 1;
141
142    let s = imms & levels;
143    let r = immr & levels;
144
145    let diff = s.wrapping_sub(r);
146
147    let esize = (1u64 << len) as u32;
148    let d = diff & levels;
149    println!("d: {}", d);
150
151    println!("s: {}, 1 << (s + 1): {}", s, (1u64 << (s + 1)) - 1);
152    let welem = (1u64 << (s + 1)) - 1;
153    let telem = (1u64 << (d + 1)) - 1;
154    println!("welem telem {} {}", welem, telem);
155    println!("esize: {}", esize);
156
157    let wmask = replicate(ror(welem, r as u32, esize), esize, M as u32);
158    let tmask = replicate(telem, esize as u32, M as u32);
159
160    (wmask, tmask)
161}
